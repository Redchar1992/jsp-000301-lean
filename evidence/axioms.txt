def JSP301.Powerful : ℕ → Prop :=
fun n => 0 < n ∧ ∀ (p : ℕ), Nat.Prime p → p ∣ n → p ^ 2 ∣ n
def JSP301.Claim : Prop :=
∀ (n : ℕ), JSP301.Powerful n → JSP301.Powerful (n + 1) → IsSquare n ∨ IsSquare (n + 1)
JSP301.counterexample : ∃ n, JSP301.Powerful n ∧ JSP301.Powerful (n + 1) ∧ ¬IsSquare n ∧ ¬IsSquare (n + 1)
